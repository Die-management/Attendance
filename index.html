// --- Enhanced Audio Beep & Mobile Vibration Feedback --- let audioCtx = null; function getAudioContext() { if (!audioCtx) { const AudioContextClass = window.AudioContext || window.webkitAudioContext; if (AudioContextClass) audioCtx = new AudioContextClass(); } if (audioCtx && audioCtx.state === 'suspended') { audioCtx.resume().catch(() => {}); } return audioCtx; } function triggerScanFeedback(isWarning = false) { // 1. Mobile Device Haptic Vibration Feedback if ("vibrate" in navigator) { try { if (isWarning) { navigator.vibrate(400); // Long single vibration for cooldown/warning } else { navigator.vibrate([150, 50, 150]); // Distinct double pulse for success } } catch (e) { console.warn('Vibration API restricted:', e); } } // 2. Loud Audio Beep Synthesizer (Web Audio API) try { const ctx = getAudioContext(); if (!ctx) return; const osc = ctx.createOscillator(); const gain = ctx.createGain(); if (isWarning) { // Low Warning Tone (220Hz Buzz) osc.type = 'sawtooth'; osc.frequency.setValueAtTime(220, ctx.currentTime); gain.gain.setValueAtTime(0.7, ctx.currentTime); // High volume gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.35); osc.connect(gain); gain.connect(ctx.destination); osc.start(ctx.currentTime); osc.stop(ctx.currentTime + 0.35); } else { // Loud High-Pitch Dual Tone (1046.5 Hz - C6 Pitch) osc.type = 'sine'; osc.frequency.setValueAtTime(1046.5, ctx.currentTime); osc.frequency.setValueAtTime(1318.5, ctx.currentTime + 0.08); // Arpeggio step to E6 gain.gain.setValueAtTime(0.85, ctx.currentTime); // Extra Loud Volume gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.2); osc.connect(gain); gain.connect(ctx.destination); osc.start(ctx.currentTime); osc.stop(ctx.currentTime + 0.2); } } catch (e) { console.warn('Loud audio feedback unavailable:', e); } } // --- Updated Scan Processing Function --- function processBarcodeScan(barcodeId) { const cleanId = barcodeId.trim(); if (!cleanId) return; const now = Date.now(); // Check 5-Minute Cooldown Window if (lastScanTimes[cleanId]) { const elapsed = now - lastScanTimes[cleanId]; if (elapsed < SCAN_COOLDOWN_MS) { const remainingSecs = Math.ceil((SCAN_COOLDOWN_MS - elapsed) / 1000); const remainingMins = Math.ceil(remainingSecs / 60); // Trigger Warning Beep & Long Vibration triggerScanFeedback(true); showAlert(`⚠️ Duplicate scan ignored! Please wait ~${remainingMins} min(s) before scanning worker ${cleanId} again.`, 'warning'); return; } } // Successful Scan -> Register Timestamp & Trigger Loud Beep + Double Vibration lastScanTimes[cleanId] = now; triggerScanFeedback(false); let worker = workersDirectory[cleanId]; if (!worker) { worker = { id: cleanId, name: `Worker ${cleanId}`, dept: 'General', type: 'Regular', agency: '' }; workersDirectory[cleanId] = worker; saveToCloud(); } const todayStr = getTodayDateString(); const currentTime = getCurrentTimeString(); let logIndex = attendanceLogs.findIndex(l => l.workerId === cleanId && l.date === todayStr); if (logIndex === -1) { const newLog = { id: 'log_' + Date.now(), workerId: worker.id, name: worker.name, dept: worker.dept || 'General', type: worker.type || 'Regular', agency: worker.agency || '', date: todayStr, inTime: currentTime, outTime: '', totalHours: '0.00', otHours: '0.00', loggedBy: currentUser ? currentUser.name : 'System', status: 'IN' }; attendanceLogs.unshift(newLog); showAlert(`✅ IN Punch logged for ${worker.name} at ${currentTime}`, 'success'); } else { let log = attendanceLogs[logIndex]; if (!log.outTime) { log.outTime = currentTime; log.status = 'OUT'; calculateLogHours(log); showAlert(`🚪 OUT Punch logged for ${worker.name} at ${currentTime} (Hrs: ${log.totalHours})`, 'purple'); } else { log.outTime = currentTime; calculateLogHours(log); showAlert(`🔄 OUT Time updated for ${worker.name} to ${currentTime}`, 'info'); } } saveToCloud(); renderDailyTable(); }
